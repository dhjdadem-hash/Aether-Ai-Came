    private fun runFullCameraDiagnostics() {
        val sb = StringBuilder()
        sb.append("=== AETHER CAMERA AI HARDWARE DIAGNOSTICS ===\n\n")

        try {
            val cameraIds = cameraManager.cameraIdList
            sb.append("عدد الكاميرات المكتشفة في النظام: ${cameraIds.size}\n")

            for (id in cameraIds) {
                val chars = cameraManager.getCameraCharacteristics(id)
                val facing = chars.get(CameraCharacteristics.LENS_FACING)
                val facingStr = when (facing) {
                    CameraCharacteristics.LENS_FACING_BACK -> "خلفية الرئيسية (BACK)"
                    CameraCharacteristics.LENS_FACING_FRONT -> "أمامية (FRONT)"
                    else -> "مستشعر إضافي"
                }

                val focalLengths = chars.get(CameraCharacteristics.LENS_INFO_AVAILABLE_FOCAL_LENGTHS)
                val maxZoom = chars.get(CameraCharacteristics.SCALER_AVAILABLE_MAX_DIGITAL_ZOOM) ?: 1.0f

                sb.append("\n📷 الكاميرا [$id] - $facingStr:\n")
                sb.append(" - Focal Lengths: ${focalLengths?.joinToString()}\n")
                sb.append(" - Max Digital Zoom Factor: ${maxZoom}x\n")

                val map = chars.get(CameraCharacteristics.SCALER_STREAM_CONFIGURATION_MAP)
                val sizes = map?.getOutputSizes(ImageFormat.JPEG)
                if (sizes != null) {
                    val maxRes = sizes.maxByOrNull { it.width * it.height }
                    if (maxRes != null) {
                        val mp = (maxRes.width * maxRes.height) / 1_000_000.0
                        sb.append(" - أعلى دقة التقاط للصور: ${maxRes.width}x${maxRes.height} (~%.1f MP)\n".format(mp))
                        
                        if (mp <= 2.5 && facing == CameraCharacteristics.LENS_FACING_BACK) {
                            secondary2MPCameraId = id
                            sb.append(" 📌 تم اكتشاف الكاميرا الثانية 2MP.\n")
                        }
                    }
                }
            }

            sb.append("\n=== تحليل مشكلة 0.8x والألوان ===\n")
            sb.append("⚠️ التشخيص: مستشعرات 2MP في أغلب الأجهزة تعمل كعدسات عمق (Depth) أو ماكرو بتركيز ثابت بدون مصفوفة ألوان متكاملة، مما يسبب الألوان الرمادية الباهتة.\n")
            sb.append("✅ إجراء AETHER ULTRA: تم تفعيل محرك AI Color Engine لتصحيح النغمة اللونية تلقائياً وحماية ألوان البشرة والطبيعة من اللون الرمادي.\n")

        } catch (e: Exception) {
            sb.append("خطأ الفحص: ${e.message}\n")
        }

        cameraDiagnosticReport = sb.toString()
        txtDiagnosticOutput.text = cameraDiagnosticReport
    }

    private fun applyZoom(zoom: Float, selectedView: TextView) {
        currentZoomRatio = zoom
        resetZoomButtonsUI()
        selectedView.setTextColor(Color.parseColor("#00E5FF"))

        if (captureRequestBuilder != null && cameraDevice != null) {
            try {
                val chars = cameraManager.getCameraCharacteristics(primaryCameraId)
                val activeArray = chars.get(CameraCharacteristics.SENSOR_INFO_ACTIVE_ARRAY_SIZE)
                
                if (activeArray != null) {
                    val effZoom = if (zoom < 1.0f) 1.0f else zoom
                    val cropW = (activeArray.width() / effZoom).toInt()
                    val cropH = (activeArray.height() / effZoom).toInt()
                    val left = (activeArray.width() - cropW) / 2
                    val top = (activeArray.height() - cropH) / 2

                    val zoomRect = Rect(left, top, left + cropW, top + cropH)
                    captureRequestBuilder?.set(CaptureRequest.SCALER_CROP_REGION, zoomRect)
                    captureSession?.setRepeatingRequest(captureRequestBuilder!!.build(), null, null)
                }
            } catch (e: Exception) {
                Log.e(TAG, "Zoom error: ${e.message}")
            }
        }
    }

    private val surfaceTextureListener = object : TextureView.SurfaceTextureListener {
        override fun onSurfaceTextureAvailable(s: SurfaceTexture, w: Int, h: Int) { openCameraHardware() }

    private fun openCameraHardware() {
        try {
            if (ActivityCompat.checkSelfPermission(this, Manifest.permission.CAMERA) != PackageManager.PERMISSION_GRANTED) return

            imageReader = ImageReader.newInstance(1920, 1080, ImageFormat.JPEG, 2).apply {
                setOnImageAvailableListener(imageAvailableListener, null)
            }

            cameraManager.openCamera(primaryCameraId, object : CameraDevice.StateCallback() {
                override fun onOpened(camera: CameraDevice) {
                    cameraDevice = camera
                    startLivePreviewSession()
                }

                override fun onDisconnected(camera: CameraDevice) {
                    cameraDevice?.close()
                    cameraDevice = null
                }

                override fun onError(camera: CameraDevice, error: Int) {
                    cameraDevice?.close()
                    cameraDevice = null
                }
            }, null)

        } catch (e: Exception) {
            Log.e(TAG, "Error opening camera: ${e.message}")
        }
    }

    private fun startLivePreviewSession() {
        try {
            val texture = textureView.surfaceTexture ?: return
            texture.setDefaultBufferSize(1920, 1080)
            val surface = Surface(texture)

            captureRequestBuilder = cameraDevice?.createCaptureRequest(CameraDevice.TEMPLATE_PREVIEW)?.apply {
                addTarget(surface)
                set(CaptureRequest.CONTROL_VIDEO_STABILIZATION_MODE, CaptureRequest.CONTROL_VIDEO_STABILIZATION_MODE_ON)
                set(CaptureRequest.CONTROL_AF_MODE, CaptureRequest.CONTROL_AF_MODE_CONTINUOUS_PICTURE)
            }

            cameraDevice?.createCaptureSession(listOf(surface, imageReader?.surface), object : CameraCaptureSession.StateCallback() {
                override fun onConfigured(session: CameraCaptureSession) {
                    captureSession = session
                    captureRequestBuilder?.let { captureSession?.setRepeatingRequest(it.build(), null, null) }
                }

                override fun onConfigureFailed(session: CameraCaptureSession) {
                    Toast.makeText(this@MainActivity, "فشل بدء المعاينة", Toast.LENGTH_SHORT).show()
                }
            }, null)
        } catch (e: Exception) {
            Log.e(TAG, "Preview Error: ${e.message}")
        }
    }

    private fun captureStillImageWithAI() {
        try {
            val captureBuilder = cameraDevice?.createCaptureRequest(CameraDevice.TEMPLATE_STILL_CAPTURE) ?: return
            captureBuilder.addTarget(imageReader!!.surface)
            captureBuilder.set(CaptureRequest.CONTROL_AF_MODE, CaptureRequest.CONTROL_AF_MODE_CONTINUOUS_PICTURE)

            captureSession?.capture(captureBuilder.build(), object : CameraCaptureSession.CaptureCallback() {
                override fun onCaptureCompleted(session: CameraCaptureSession, request: CaptureRequest, result: TotalCaptureResult) {
                    runOnUiThread { Toast.makeText(this@MainActivity, "جاري معالجة الصورة بـ AI ULTRA...", Toast.LENGTH_SHORT).show() }
                }
            }, null)
        } catch (e: Exception) {
            Log.e(TAG, "Capture error: ${e.message}")
        }
    }

    private val imageAvailableListener = ImageReader.OnImageAvailableListener { reader ->
        val image = reader.acquireNextImage() ?: return@OnImageAvailableListener
        val buffer = image.planes[0].buffer
        val bytes = ByteArray(buffer.remaining())
        buffer.get(bytes)
        image.close()

        Thread {
            var bitmap = BitmapFactory.decodeByteArray(bytes, 0, bytes.size)
            if (isAiColorEnabled || currentZoomRatio < 1.0f) {
                bitmap = applyRealAIColorCorrection(bitmap)
            }
            saveBitmapToGallery(bitmap)
        }.start()
    }

    private fun applyRealAIColorCorrection(source: Bitmap): Bitmap {
        val mutableBitmap = source.copy(Bitmap.Config.ARGB_8888, true)
        val canvas = Canvas(mutableBitmap)

        val paint = Paint().apply {
            val cm = ColorMatrix().apply {
                setSaturation(1.30f)
                postConcat(ColorMatrix(floatArrayOf(
                    1.08f, 0f, 0f, 0f, 12f,
                    0f, 1.04f, 0f, 0f, 6f,
                    0f, 0f, 1.10f, 0f, 8f,
                    0f, 0f, 0f, 1f, 0f
                )))
            }
            colorFilter = ColorMatrixColorFilter(cm)
        }

        canvas.drawBitmap(mutableBitmap, 0f, 0f, paint)
        return mutableBitmap
    }

    private fun saveBitmapToGallery(bitmap: Bitmap) {
        val filename = "AETHER_${SimpleDateFormat("yyyyMMdd_HHmmss", Locale.US).format(Date())}.jpg"
        val cv = ContentValues().apply {
            put(MediaStore.MediaColumns.DISPLAY_NAME, filename)
            put(MediaStore.MediaColumns.MIME_TYPE, "image/jpeg")
            put(MediaStore.MediaColumns.RELATIVE_PATH, Environment.DIRECTORY_DCIM + "/AetherCamera")
        }

        val uri: Uri? = contentResolver.insert(MediaStore.Images.Media.EXTERNAL_CONTENT_URI, cv)
        if (uri != null) {
            contentResolver.openOutputStream(uri)?.use { out ->
                bitmap.compress(Bitmap.CompressFormat.JPEG, 95, out)
                runOnUiThread { Toast.makeText(this, "تم حفظ الصورة بنجاح!", Toast.LENGTH_SHORT).show() }
            }
        }
    }

    private fun startSafeVideoRecording() {
        try {
            val videoFile = File(getExternalFilesDir(Environment.DIRECTORY_MOVIES), "VID_AETHER_${System.currentTimeMillis()}.mp4")
            currentVideoFilePath = videoFile.absolutePath

            mediaRecorder = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) MediaRecorder(this) else MediaRecorder()
            mediaRecorder?.apply {
                setAudioSource(MediaRecorder.AudioSource.MIC)
                setVideoSource(MediaRecorder.VideoSource.SURFACE)
                setOutputFormat(MediaRecorder.OutputFormat.MPEG_4)
                setOutputFile(currentVideoFilePath)
                setVideoEncodingBitRate(12000000)
                setVideoFrameRate(30)
                setVideoSize(1920, 1080)
                setVideoEncoder(MediaRecorder.VideoEncoder.H264)
                setAudioEncoder(MediaRecorder.AudioEncoder.AAC)
                prepare()
            }

            isRecordingVideo = true
            btnRecord.setBackgroundDrawable(createGlassPillDrawable("#FF1744"))
            Toast.makeText(this, "بدأ تسجيل الفيديو بتثبيت الحركة الحقيقي...", Toast.LENGTH_SHORT).show()
        } catch (e: Exception) {
            Toast.makeText(this, "فشل بدء التسجيل: ${e.message}", Toast.LENGTH_SHORT).show()
        }
    }

    private fun stopSafeVideoRecording() {
        if (!isRecordingVideo) return
        try {
            mediaRecorder?.apply {
                stop()
                reset()
                release()
            }
            mediaRecorder = null
            isRecordingVideo = false
            btnRecord.setBackgroundDrawable(createRecordDrawable())
            Toast.makeText(this, "تم حفظ الفيديو بنجاح دون أخطاء!", Toast.LENGTH_LONG).show()
        } catch (e: Exception) {
            Toast.makeText(this, "خطأ أثناء إنهاء التسجيل", Toast.LENGTH_SHORT).show()
        }
    }

    private fun setCaptureMode(mode: CaptureMode, activeTv: TextView) {
        currentMode = mode
        resetModesUI()
        activeTv.setTextColor(Color.parseColor("#00E5FF"))

        if (mode == CaptureMode.VIDEO) {
            btnShutter.visibility = View.GONE
            btnRecord.visibility = View.VISIBLE
        } else {
            btnShutter.visibility = View.VISIBLE
            btnRecord.visibility = View.GONE
        }
    }

    private fun resetModesUI() {
        txtModePhoto.setTextColor(Color.WHITE)
        txtModeVideo.setTextColor(Color.WHITE)
        txtModeNight.setTextColor(Color.WHITE)
        txtModePortrait.setTextColor(Color.WHITE)
        txtModePro.setTextColor(Color.WHITE)
        txtModeAstro.setTextColor(Color.WHITE)
    }

    private fun createZoomButton(txt: String, isSelected: Boolean = false, isSpecial: Boolean = false): TextView {
        return TextView(this).apply {
            text = txt
            setTextColor(if (isSelected || isSpecial) Color.parseColor("#00E5FF") else Color.WHITE)
            textSize = 11f
            setTypeface(null, Typeface.BOLD)
            setPadding(20, 8, 20, 8)
        }
    }

    private fun createModeText(txt: String, isActive: Boolean = false): TextView {
        return TextView(this).apply {
            text = txt
            setTextColor(if (isActive) Color.parseColor("#00E5FF") else Color.WHITE)
            textSize = 12f
            setTypeface(null, Typeface.BOLD)
            setPadding(25, 0, 25, 0)
        }
    }

    private fun createGlassPillDrawable(colorHex: String = "#33FFFFFF"): android.graphics.drawable.GradientDrawable {
        return android.graphics.drawable.GradientDrawable().apply {
            setColor(Color.parseColor(colorHex))
            cornerRadius = 80f
            setStroke(2, Color.parseColor("#44FFFFFF"))
        }
    }

    private fun createShutterDrawable(): android.graphics.drawable.GradientDrawable {
        return android.graphics.drawable.GradientDrawable().apply {
            shape = android.graphics.drawable.GradientDrawable.OVAL
            setColor(Color.WHITE)
            setStroke(10, Color.parseColor("#88FFFFFF"))
        }
    }

    private fun createRecordDrawable(): android.graphics.drawable.GradientDrawable {
        return android.graphics.drawable.GradientDrawable().apply {
            shape = android.graphics.drawable.GradientDrawable.OVAL
            setColor(Color.parseColor("#FF1744"))
            setStroke(10, Color.parseColor("#88FFFFFF"))
        }
    }

    private fun checkPermissionsGranted() = REQUIRED_PERMISSIONS.all {
        ContextCompat.checkSelfPermission(baseContext, it) == PackageManager.PERMISSION_GRANTED
    }
